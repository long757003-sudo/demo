```javascript
if (typeof window._daCandidates === 'undefined') window._daCandidates = [];

function _daSearch() {
  var kw = (document.getElementById('da-search').value || '').trim();
  var result = document.getElementById('da-search-result');
  if (!kw) { result.innerHTML = '<div class="empty">请输入关键词</div>'; return; }
  var matches = DATA.demandUsers.filter(function(u) {
    return u.name.indexOf(kw) >= 0 || u.empNo.indexOf(kw) >= 0;
  }).filter(function(u) {
    // 仅显示 project-manager 角色可指派
    return u.roles && u.roles.indexOf('project-manager') >= 0;
  });
  if (matches.length === 0) {
    result.innerHTML = '<div class="empty">未找到匹配人员</div>'; return;
  }
  result.innerHTML = matches.map(function(u) {
    return '<div class="user-item">' +
      '<span>' + u.name + ' ' + u.empNo + '</span>' +
      '<button class="btn btn-sm" onclick="_daAddCandidate(\'' + u.id + '\')">加入备选</button>' +
      '</div>';
  }).join('');
}

function _daAddCandidate(uid) {
  var u = DATA.demandUsers.find(function(x){return x.id===uid;});
  if (!u) return;
  if (window._daCandidates.some(function(c){return c.id===uid;})) {
    toast('该人员已在备选列表中', 'warning'); return;
  }
  window._daCandidates.push(u);
  toast('已加入备选列表', 'success');
  renderView('demand-assign');
}

function _daRemoveCandidate(uid) {
  window._daCandidates = window._daCandidates.filter(function(c){return c.id!==uid;});
  renderView('demand-assign');
}

function _daSubmit() {
  if (!window._daCandidates || window._daCandidates.length === 0) {
    toast('请至少指派 1 名填报人', 'error'); return;
  }
  var body = '<p>即将指派以下 ' + window._daCandidates.length + ' 名填报人：</p>' +
    '<ul style="margin:8px 0;padding-left:20px">' +
    window._daCandidates.map(function(u){return '<li>'+u.name+' '+u.empNo+'</li>';}).join('') +
    '</ul><p>确认后将发送指派通知。</p>';
  showModal('确认指派填报人', body,
    '<button class="btn" onclick="closeModal()">取消</button>' +
    '<button class="btn btn-primary" onclick="_daConfirmSubmit()">确认提交</button>');
}

function _daConfirmSubmit() {
  logOperation('demand-assign', 'assign', window._daCandidates.length + '人');
  toast('已指派 ' + window._daCandidates.length + ' 名填报人', 'success');
  closeModal();
  window._daCandidates = [];
  navigate('demand-list');
}
```

```javascript
if (typeof window._demandFormData === 'undefined') window._demandFormData = {};
if (typeof window._demandFormStep === 'undefined') window._demandFormStep = 0;

function _dfCaptureStep() {
  // 读取当前 step 所有字段写入 window._demandFormData
  // step 0: name, contact, budget
  // step 1: summary, desc, priority
  // step 2: tech, tags (勾选的 checkbox value 数组), relation
  // step 3: remark (文件 input 不持久化)
  var step = window._demandFormStep;
  var d = window._demandFormData;
  if (step === 0) {
    d.name = _val('df-name'); d.contact = _val('df-contact'); d.budget = _val('df-budget');
  } else if (step === 1) {
    d.summary = _val('df-summary'); d.desc = _val('df-desc');
    var pr = document.querySelector('input[name="df-priority"]:checked');
    d.priority = pr ? pr.value : '';
  } else if (step === 2) {
    d.tech = _val('df-tech');
    d.tags = Array.prototype.map.call(
      document.querySelectorAll('.df-tag:checked'),
      function(el){return {id:el.value, category:el.dataset.cat, name:el.dataset.name};}
    );
    d.relation = _val('df-relation');
  } else if (step === 3) {
    d.remark = _val('df-remark');
  }
}

function _val(id) { var el = document.getElementById(id); return el ? el.value : ''; }

function _dfSaveDraft() {
  _dfCaptureStep();
  saveDraft('demand-fill', window._demandFormData);
  toast('草稿已保存', 'success');
}

function _dfNextStep() {
  _dfCaptureStep();
  // Step 0 必填校验
  if (window._demandFormStep === 0) {
    var d = window._demandFormData;
    if (!d.name) { toast('请填写「项目名称」','error'); return; }
    if (!d.contact) { toast('请填写「联系方式」','error'); return; }
    if (!d.budget) { toast('请填写「预期投入金额」','error'); return; }
  }
  if (window._demandFormStep === 1) {
    var d = window._demandFormData;
    if (!d.summary) { toast('请填写「摘要」','error'); return; }
    if (!d.desc) { toast('请填写「功能描述」','error'); return; }
  }
  window._demandFormStep++;
  renderView('demand-fill');
}

function _dfPrevStep() {
  _dfCaptureStep();
  if (window._demandFormStep > 0) window._demandFormStep--;
  renderView('demand-fill');
}

function _dfSubmit() {
  _dfCaptureStep();
  var d = window._demandFormData;
  if (!d.name || !d.contact || !d.budget || !d.summary || !d.desc) {
    toast('请完整填写前序步骤的必填字段', 'error'); return;
  }
  var demand = {
    id: 'dm-' + Date.now(),
    name: d.name,
    unitId: (DATA.demandUsers.find(function(u){return u.roles && u.roles.indexOf('project-manager')>=0;})||{}).unitId || 'unit-edu',
    collectionId: (DATA.collectionPlans[0]||{}).id || '',
    submitterId: 'u-wangyifan',
    submitter: '王一凡 50240014',
    summary: d.summary,
    description: d.desc,
    budget: Number(d.budget) || 0,
    priority: d.priority || '中',
    tech: d.tech || '',
    tags: d.tags || [],
    relation: d.relation || '',
    remark: d.remark || '',
    status: 'submitted',
    sortOrder: 999,
    createdAt: new Date().toISOString().slice(0,10)
  };
  if (!Array.isArray(DATA.demands)) DATA.demands = [];
  DATA.demands.unshift(demand);
  logOperation('demand-fill', 'submit', demand.name);
  clearDraft('demand-fill');
  window._demandFormData = {};
  window._demandFormStep = 0;
  toast('需求已提交，等待单位系统管理员处理', 'success');
  // D-06：提交后跳转推进流程演示（这里跳 demand-list 展示多份需求列表）
  navigate('demand-list');
}

function _dfUpdateTypeBadge() {
  var v = Number(_val('df-budget')) || 0;
  var el = document.getElementById('df-budget-badge');
  if (el && typeof projectTypeTag === 'function' && typeof budgetToType === 'function') {
    el.innerHTML = projectTypeTag(budgetToType(v));
  }
}

function _dfStartNew() {
  window._demandFormData = {};
  window._demandFormStep = 0;
  clearDraft('demand-fill');
  navigate('demand-fill');
}
```
